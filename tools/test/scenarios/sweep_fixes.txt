summon tadpole 201 -61 201 {Tags:["c60t"],Age:23900}

execute positioned 201 -61 201 run mobstacker stats 8
execute as @e[type=frog,x=190,y=-64,z=190,dx=25,dy=10,dz=25] run data get entity @s StackData.StackSize
